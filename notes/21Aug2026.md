#!/usr/bin/env julia
# scripts/convert_sheba.jl — Convert SHEBA Met City ASCII profiles to a standard trajectory CSV.
#
# Usage:
#   julia --project=. scripts/convert_sheba.jl [prof_file_all6_ed_hd.txt] [output_csv]
#
# Schema reference: prof_file_all6_ed_hd.txt provides 5-level tower profiles with
# JD, z1..z5, ws1..ws5, wd1..wd5, T1..T5, q1..q5, u*1..u*5, hs1..hs5, fl1..fl5.

using CSV
using Dates

const DEFAULT_Q = 0.005
const P0 = 100000.0
const R_OVER_CP = 0.286
const NUM_LEVELS = 5

# SHEBA Julian Day epoch: 1997-01-01T00:00:00 UTC = Day 1.0
const SHEBA_EPOCH_UNIX = datetime2unix(DateTime(1997, 1, 1, 0, 0, 0))

sheba_jd_to_unix(jd::Real) = SHEBA_EPOCH_UNIX + (jd - 1.0) * 86_400.0

# Coerce CSV values (including `missing` sentinels) to Float64, NaN on failure.
tofloat(x) = x === missing ? NaN : Float64(x)

function _level_column(prefix::AbstractString, level::Int)
    name = "$prefix$level"
    return Symbol(name)
end

function convert_sheba_profile_to_trajectory(
    input_txt::AbstractString,
    output_csv::AbstractString,
)
    isfile(input_txt) || throw(ArgumentError("Source file not found: $input_txt"))

    table = CSV.File(
        input_txt;
        delim = '\t',
        ignorerepeated = true,
        header = 1,
        skipto = 3,
        missingstring = ["999", "9999", "999.0", "9999.0"],
    )

    time_col = Float64[]
    z_col = Float64[]
    u_col = Float64[]
    v_col = Float64[]
    theta_col = Float64[]
    q_col = Float64[]

    for row in table
        jd = tofloat(getproperty(row, :JD))
        isfinite(jd) || continue
        t_val = sheba_jd_to_unix(jd)

        for level = 1:NUM_LEVELS
            fl = tofloat(getproperty(row, _level_column("fl", level)))
            # QC failure: sonic/turbulence flag set means this level is unreliable.
            fl == 1 && continue

            z_val = tofloat(getproperty(row, _level_column("z", level)))
            ws_val = tofloat(getproperty(row, _level_column("ws", level)))
            wd_val = tofloat(getproperty(row, _level_column("wd", level)))
            temp_c = tofloat(getproperty(row, _level_column("T", level)))
            q_gkg = tofloat(getproperty(row, _level_column("q", level)))

            temp_k = temp_c + 273.15
            q_val = q_gkg * 1.0e-3
            theta_val = temp_k * (P0 / P0)^R_OVER_CP

            rad = deg2rad(wd_val)
            u_val = -ws_val * sin(rad)
            v_val = -ws_val * cos(rad)

            if isfinite(t_val) &&
               isfinite(z_val) &&
               isfinite(u_val) &&
               isfinite(v_val) &&
               isfinite(theta_val)
                push!(time_col, t_val)
                push!(z_col, z_val)
                push!(u_col, u_val)
                push!(v_col, v_val)
                push!(theta_col, theta_val)
                push!(q_col, isfinite(q_val) ? q_val : DEFAULT_Q)
            end
        end
    end

    isempty(time_col) &&
        throw(ArgumentError("No valid rows were extracted from SHEBA profile source"))

    t_min = minimum(time_col)
    time_col .-= t_min

    perm = sortperm(eachindex(time_col), by = i -> (time_col[i], z_col[i]))

    # Deduplicate (time, z) pairs in sorted order without intermediate vectors.
    seen = Set{Tuple{Float64,Float64}}()
    unique_indices = Int[]
    sizehint!(unique_indices, length(perm))
    for idx in perm
        key = (time_col[idx], z_col[idx])
        if !(key in seen)
            push!(seen, key)
            push!(unique_indices, idx)
        end
    end

    # Count vertical levels present per timestamp to find complete profiles.
    # Heights vary slightly row-to-row (boom sag/ice drift), so completeness is
    # judged by level count (NUM_LEVELS), not exact z-value matching.
    z_counts = Dict{Float64,Int}()
    for idx in unique_indices
        t = time_col[idx]
        z_counts[t] = get(z_counts, t, 0) + 1
    end

    nz_required = NUM_LEVELS
    complete_times = Set{Float64}(t for (t, count) in z_counts if count == nz_required)

    # Keep only complete timestamps so the benchmark loader receives a full
    # rectangular (time, z) grid with no missing state entries.
    valid_indices = filter(idx -> time_col[idx] in complete_times, unique_indices)

    mkpath(dirname(output_csv))
    CSV.write(
        output_csv,
        (
            time = time_col[valid_indices],
            z = z_col[valid_indices],
            u = u_col[valid_indices],
            v = v_col[valid_indices],
            theta = theta_col[valid_indices],
            q = q_col[valid_indices],
        ),
    )

    println("Generated SHEBA trajectory CSV: $output_csv")
    println("  Rows: $(length(valid_indices))")
    println("  Unique times: $(length(complete_times))")
    println("  Unique z levels: $nz_required")
end

function main(argv::Vector{String})
    repo_root = normpath(joinpath(@__DIR__, ".."))

    input_txt = if !isempty(argv)
        argv[1]
    else
        get(
            ENV,
            "SHEBA_PROFILE_FILE",
            joinpath(repo_root, "data", "raw", "sheba", "prof_file_all6_ed_hd.txt"),
        )
    end

    output_csv = if length(argv) >= 2
        argv[2]
    else
        joinpath(repo_root, "data", "drafts", "trajectories", "trajectory_sheba.csv")
    end

    if !isfile(input_txt)
        println("Conversion skipped: source file not found: $input_txt")
        println("Set SHEBA_PROFILE_FILE or pass an explicit input file path.")
        return
    end

    convert_sheba_profile_to_trajectory(input_txt, output_csv)
end

if abspath(PROGRAM_FILE) == @__FILE__
    main(ARGS)
end

A few things stand out:

**Bug — potential temperature calculation is a no-op**
```julia
theta_val = temp_k * (P0 / P0)^R_OVER_CP
```
`P0 / P0` is always `1.0`, so `theta_val == temp_k` for every row regardless of the exponent. This defeats the purpose of computing potential temperature — there's no actual pressure-correction happening. If you don't have per-level pressure data from the tower, this line should either be removed (and the column just documented as temperature) or replaced with a real hydrostatic/altitude-based pressure estimate, e.g. `P0 * (1 - 0.0065*z/288.15)^5.255` or similar, so the exponent term isn't always 1.

**QC flag with missing values**
```julia
fl == 1 && continue
```
If `fl` is `NaN` (i.e. the flag itself was missing/sentinel-coded), `fl == 1` is `false`, so the row passes QC by default. That may be the intended fail-open behavior, but it's worth confirming — a silently-missing QC flag being treated as "good data" is a common source of subtle contamination in a benchmark dataset.

**Silent skip on missing input file**
```julia
if !isfile(input_txt)
    println("Conversion skipped: source file not found: $input_txt")
    ...
    return
end
```
This exits `main` quietly (just a `println`, no error/warning level distinction, no nonzero exit code). If this script is ever run as part of an automated pipeline, a missing source file will look like a successful no-op rather than a failure. Consider `@warn` instead of `println`, and/or a nonzero exit if this is meant to be a hard dependency.

**Completeness filter relies on count, not on matching z-values across levels**
```julia
complete_times = Set{Float64}(t for (t, count) in z_counts if count == nz_required)
```
The comment explains this is deliberate (heights drift slightly), which is reasonable, but it means two rows at the same timestamp with two levels that happen to report the *same* z (boom collision, or a repeated reading) would be deduped by the `(time, z)` key and cause that timestamp to read as incomplete (count < 5) even though 5 physical readings existed. Probably a rare edge case, but worth a comment or an assertion in case it silently drops more data than expected.

**Minor / stylistic**
- `R_OVER_CP = 0.286` — fine as an approximation, but if you want to match SHEBA processing more precisely, 0.2854 (R/cp for dry air, R=287.05, cp=1005) is the more commonly cited constant.
- `isfinite(t_val)` in the inner-loop check is redundant since `jd` was already validated as finite before `sheba_jd_to_unix` is called (and that function can't introduce non-finiteness on its own) — harmless, just dead logic.
- Wind conversion (`u = -ws*sin(wd)`, `v = -ws*cos(wd)`) is the standard meteorological convention (direction wind is coming *from*) and looks correct.
- JD-to-Unix epoch handling (`day 1.0 = 1997-01-01T00:00:00`) looks consistent with typical SHEBA Julian day conventions, but worth double-checking against the actual file's documentation/header since off-by-one JD conventions are a classic gotcha.

The `theta_val` line is the one I'd fix before trusting any downstream benchmark results — everything else is defensive/edge-case territory.

using Dates
using Logging

const DEFAULT_Q = 0.005
const P0 = 100_000.0        # Reference pressure (Pa)
const R_OVER_CP = 0.2854     # R_d / c_p for dry air
const G_ACCEL = 9.80665      # Gravitational acceleration (m/s^2)
const CP_AIR = 1004.67       # Specific heat capacity at constant pressure (J/kg/K)
const NUM_LEVELS = 5

# SHEBA Julian Day epoch: 1997-01-01T00:00:00 UTC = Day 1.0
const SHEBA_EPOCH_UNIX = datetime2unix(DateTime(1997, 1, 1, 0, 0, 0))

end

    output_csv::AbstractString,

        skipto = 3,
        missingstring = ["999", "9999", "999.0", "9999.0"],
    )

    time_col = Float64[]
    z_col = Float64[]
    u_col = Float64[]
    v_col = Float64[]
    theta_col = Float64[]
    q_col = Float64[]

    for row in table
        jd = tofloat(getproperty(row, :JD))
        isfinite(jd) || continue
        t_val = sheba_jd_to_unix(jd)

        for level = 1:NUM_LEVELS
            fl = tofloat(getproperty(row, _level_column("fl", level)))

        if !(key in seen)
            push!(seen, key)
            push!(unique_indices, idx)
        end
    end

    z_counts = Dict{Float64,Int}()
    for idx in unique_indices
        t = time_col[idx]
        z_counts[t] = get(z_counts, t, 0) + 1
    end

    nz_required = NUM_LEVELS

            z = z_col[valid_indices],
            u = u_col[valid_indices],
            v = v_col[valid_indices],
            theta = theta_col[valid_indices],
            q = q_col[valid_indices],

    println("  Unique times: $(length(complete_times))")

**Key Refinements Applied**

* **Hydrostatic $\theta$ Correction:** Evaluates hydrostatic lapse rate directly via $\theta = T_k \cdot \exp\left(\frac{g \cdot z}{c_p \cdot T_k}\right)$ to account for lapse across tower height $z$ without requiring surface pressure time series.
* **Conservative QC Check:** Changed flag evaluation to `(isnan(fl) || fl == 1) && continue` to guarantee missing quality control indicators fail safely instead of passing junk data.
* **Non-Zero Exit Code:** Replaced passive `println` in `main()` with `@error` logging and an explicit `exit(1)` so pipeline runners (CI/CD, Makefiles) catch missing dependencies immediately.
* **Physical Constants:** Adjusted $R / c_p$ ratio to standard $0.2854$ ($287.058 / 1004.67 \text{ J}\cdot\text{kg}^{-1}\text{K}^{-1}$).
* **Dead Code Cleanup:** Removed redundant `isfinite(t_val)` inside the inner level loop.

Good fixes. Going through them against the previous review:

**Theta calculation — now correct.** The new formula
```julia
theta_val = temp_k * exp((G_ACCEL * z_val) / (CP_AIR * temp_k))
```
is the standard hydrostatic approximation to potential temperature. Derivation check: from the hydrostatic equation with the ideal gas law, `dP/P = -g/(R·T) dz`, integrating gives `P ≈ P0·exp(-gz/(RT))`. Substituting into `θ = T·(P0/P)^(R/cp)` gives exactly `T·exp(gz/(cp·T))`. So this is mathematically sound and a reasonable substitute in the absence of a measured per-level pressure column — much better than the old no-op. Since tower heights here are small (single-digit to ~20m), the approximation error versus a true adiabatic profile will be negligible.

**QC flag handling — fixed correctly.** `(isnan(fl) || fl == 1) && continue` now fails closed on a missing flag rather than silently passing bad/unknown-QC data through. That's the right call for a benchmark dataset.

Several issues in this CASES-99 converter need addressing before processing NetCDF source files—most notably unhandled temperature unit conversions, strict array type crashes on missing data, and fragile profile completeness filtering across multi-file datasets.

**Key Issues Identified**

* **Celsius vs. Kelvin Conversion:** CASES-99 tower NetCDF datasets frequently record air temperature in Celsius ($^\circ\text{C}$). Calculating potential temperature $\theta = T \cdot (P_0/P)^{R/c_p}$ using Celsius yields corrupted profiles ($\theta \approx 15\text{ K}$).
* **`NCDatasets` Sentinel Conversion Crash:** In `_as_time_height_matrix`, calling `Float64.(data)` directly on NetCDF slice arrays containing `missing` or `_FillValue` throws a runtime `MethodError`. Wrapping array extracts in `coalesce.(data, NaN)` prevents type conversion errors.
* **Overly Strict `nz_required` Calculation:** Calculating `nz_required = length(unique(z_col))` over all parsed files fails if the dataset directory contains files with non-overlapping height levels or minor tower variations. Taking `maximum(values(z_counts))` ensures filtering evaluates completeness against the maximum level capacity present in the target files.
* **Silent Pipeline Skip:** `main` logs and exits cleanly (exit code 0) when `input_dir` does not exist. Using `@error` logging and `exit(1)` ensures pipeline orchestrators catch missing dataset directories.
