A dual `CMakeLists.txt` and `Makefile` setup ensures clean compilation of `sbl_column_gate.f90` across Linux (`.so`), macOS (`.dylib`), and Windows (`.dll`) targets for direct linking into WRF-ARW, MPAS-Atmosphere, or local CFFI test runners.

**CMake Configuration (`CMakeLists.txt`)**

```cmake
cmake_minimum_required(VERSION 3.18)
project(SBLColumnGate LANGUAGES Fortran C)

set(CMAKE_Fortran_STANDARD 2003)
set(CMAKE_Fortran_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

# Source files and shared target
add_library(sbl_column_gate SHARED sbl_column_gate.f90)

set_target_properties(sbl_column_gate PROPERTIES
    PREFIX "lib"
    Fortran_MODULE_DIRECTORY ${CMAKE_BINARY_DIR}/include
)

# Compiler optimization flags
target_compile_options(sbl_column_gate PRIVATE
    $<$<Fortran_COMPILER_ID:GNU>:-O3 -Wall -Wextra -std=f2003>
    $<$<Fortran_COMPILER_ID:Intel>:-O3 -std03>
)

# Install rules for headers/modules and shared library
install(TARGETS sbl_column_gate
    LIBRARY DESTINATION lib
    ARCHIVE DESTINATION lib
    RUNTIME DESTINATION bin
)
install(DIRECTORY ${CMAKE_BINARY_DIR}/include/ DESTINATION include)

```

**GNU Makefile (`Makefile`)**

```makefile
FC ?= gfortran
FFLAGS ?= -O3 -fPIC -std=f2003 -Wall -Wextra
LIB_NAME = libsbl_column_gate

ifeq ($(OS),Windows_NT)
    EXT = .dll
else ifeq ($(shell uname),Darwin)
    EXT = .dylib
else
    EXT = .so
endif

TARGET = $(LIB_NAME)$(EXT)
SRC = sbl_column_gate.f90
OBJ = $(SRC:.f90=.o)

all: $(TARGET)

$(TARGET): $(OBJ)
 $(FC) -shared -o $@ $^

%.o: %.f90
 $(FC) $(FFLAGS) -c $< -o $@

clean:
 rm -f *.o *.mod $(LIB_NAME).so $(LIB_NAME).dylib $(LIB_NAME).dll

.PHONY: all clean

```

**Compilation & Test Workflow**

* **Build Library:** Run `make` or `cmake -B build && cmake --build build` to produce `libsbl_column_gate.so` (or platform equivalent).
* **Run Cross-Language Suite:** Execute `python test_fortran_interop.py` to confirm C-struct memory alignment and bit-for-bit equivalence against the frozen Julia baseline.
* **NW3 Integration Link:** Pass `-I/path/to/include -L/path/to/lib -lsbl_column_gate` into the WRF or MPAS physics Makefile to expose `sbl_fast_gate` to the SCM module driver.
